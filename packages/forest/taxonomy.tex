\usepackage[edges]{forest}

\begin{forest}
  for tree={

    draw, rounded corners, font=\small, inner sep=2.5pt,
    anchor=west, child anchor=west, edge={thick, gray!70!black},
  },
  where level=0{rotate=90, anchor=north, child anchor=north, fill=blue!70!black, text=white, font=\small\bfseries}{},
  where level=1{fill=blue!15}{},
  where level=2{fill=blue!6}{},
  where n children=0{draw=gray!60, font=\footnotesize}{},
  [Graph neural networks

      [Spatial [GraphSAGE] [GAT]]
    ]
    [Recurrent [GGNN]]
    [Autoencoders [GAE] [VGAE]]
    [Spatial-temporal [STGCN]]
  ]
\end{forest}
