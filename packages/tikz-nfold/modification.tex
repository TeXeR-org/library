  \arrow[Rightarrow, nfold, from=U, to=D, bend right=45, shorten=6pt, "\alpha"'{name=L}]
  \arrow[Rightarrow, nfold, from=U, to=D, bend left=45, shorten=6pt, "\beta"{name=R}]
  \arrow[Rightarrow, scaling nfold=3, from=L, to=R, shorten=6pt, "\Gamma"]
