\begin{tikzcd}[row sep=large, column sep=large]

  \cdots \arrow[r] & D_{n+1} \arrow[r, "\partial'_{n+1}"'] & D_n \arrow[r, "\partial'_n"'] & D_{n-1} \arrow[r] & \cdots
