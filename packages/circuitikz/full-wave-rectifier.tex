% Name: Full-wave rectifier with a smoothing capacitor
% Description: A center-tapped transformer and two diodes that conduct on alternate half-cycles, with a capacitor across the load to smooth the output.
% Tags: rectifier, center tap, smoothing capacitor, power supply

\usepackage{circuitikz}

\begin{circuitikz}[american voltages]
  \draw (0,1.2) to[sV, l_=$v_{in}$] (0,-1.2);
  \draw (0,1.2) -- (1,1.2) to[L, mirror] (1,-1.2) -- (0,-1.2);
  \draw[thick] (1.35,1.3) -- (1.35,-1.3) (1.5,1.3) -- (1.5,-1.3);
  \draw (1.85,1.2) to[L, name=S] (1.85,-1.2);
  \draw (1.85,1.2) -- (1.85,2.2) to[D*, l=$D_1$] (3.6,2.2) -- (7.6,2.2);
  \draw (1.85,-1.2) to[D*, l_=$D_2$] (7.6,-1.2) -- (7.6,2.2);
  % The center tap is the inductor's midtap anchor.
  \draw (S.midtap) to[short, *-] (6.2,0 |- S.midtap);
  \draw (4.8,2.2) to[C, l_=$C$, *-*] (4.8,0 |- S.midtap);
  \draw (6.2,2.2) to[R, l_=$R_L$, v^=$v_{out}$, *-] (6.2,0 |- S.midtap);
