% Name: Ringed planet
% Description: A tilted planet with bands, its ring passing behind it at the top and in front of it at the bottom, among a few stars.
% Tags: planet, Saturn, space, astronomy, illustration

\begin{tikzpicture}
  \foreach \p/\s in {(-2.1,1.2)/0.06, (1.9,1.4)/0.08, (2.3,-1.1)/0.05, (-1.7,-1.3)/0.07, (0.4,1.75)/0.05}
    \fill[orange!80!yellow] \p circle (\s);
  \begin{scope}[rotate=-18]
    % Back of the ring, then the planet, then the front of the ring over it
    \draw[orange!45!brown, line width=5pt] (0,0) ellipse (2 and 0.42);
    \begin{scope}
      \clip (0,0) circle (1);
      \fill[orange!45] (-1,-1) rectangle (1,1);
      \fill[orange!25] (-1,0.25) rectangle (1,0.5) (-1,-0.15) rectangle (1,-0.05) (-1,-0.7) rectangle (1,-0.5);
    \end{scope}
    \draw[orange!50!black, thick] (0,0) circle (1);
    \draw[orange!45!brown, line width=5pt] (2,0) arc[start angle=0, end angle=-180, x radius=2, y radius=0.42];
  \end{scope}
\end{tikzpicture}
