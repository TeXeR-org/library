% Description: A daisy-like flower with eight pink petals round a yellow center, on a green stem with two leaves.

  \filldraw[fill=yellow!80!orange, draw=orange!70!black] (0,3) circle (0.36);
  \foreach \a in {0,60,...,300} \fill[orange!70!black] (0,3) ++(\a+20:0.18) circle (0.035);
