% Name: Umbrella
% Description: An open red umbrella with a scalloped edge and a hooked handle, with raindrops falling around it.
% Tags: umbrella, rain, weather, illustration

\begin{tikzpicture}[line join=round]
  % Canopy: a half circle of radius 2 whose lower edge is four arcs of radius 0.5
  \filldraw[fill=red!65!black, draw=red!35!black, thick] (-2,0) arc (180:0:2) \foreach \k in {0,...,3} { arc[start angle=0, end angle=180, radius=0.5, x radius=0.5, y radius=0.3] } -- cycle;
  \foreach \x in {-1,0,1} \draw[red!35!black, thin] (\x,0) .. controls (\x*0.6,1) and (\x*0.2,1.8) .. (0,2);
  \fill[red!35!black] (0,2) rectangle ++(0.06,0.25);
  % Shaft and J hook
  \draw[gray!40!black, line width=2pt, line cap=round] (0,0) -- (0,-2.2) arc (180:360:0.3) -- ++(0,0.15);
  \foreach \x/\y in {-2.6/0.8, -2.2/-0.6, 2.4/1.4, 2.7/-0.2, -1.4/-1.6, 1.6/-1.3, 0.9/2.9, -1.2/2.8}
    \fill[cyan!60!blue] (\x,\y) .. controls ++(0.1,-0.12) and ++(0.1,0) .. ++(0,-0.25) .. controls ++(-0.1,0) and ++(-0.1,-0.12) .. ++(0,0.25);
\end{tikzpicture}
