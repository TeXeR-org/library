% Name: Cat face
% Description: A friendly orange cat's face with pointed ears, green eyes, a pink nose, and whiskers.
% Tags: cat, animal, face, illustration

\begin{tikzpicture}[line join=round]
  % Ears behind the head
  \foreach \s in {-1,1} {
    \filldraw[fill=orange!60, draw=orange!50!black, thick] (\s*0.55,0.75) -- (\s*1.05,1.75) -- (\s*1.15,0.4) -- cycle;
    \fill[pink!70] (\s*0.68,0.75) -- (\s*1,1.45) -- (\s*1.05,0.65) -- cycle;
  }
  \filldraw[fill=orange!60, draw=orange!50!black, thick] (0,0) ellipse (1.35 and 1.15);
  % Eyes: green with slit pupils and a highlight
  \foreach \s in {-1,1} {
    \filldraw[fill=green!55!black!70, draw=black, thick] (\s*0.5,0.25) ellipse (0.24 and 0.3);
    \fill (\s*0.5,0.25) ellipse (0.07 and 0.24);
    \fill[white] (\s*0.5+0.08,0.38) circle (0.045);
  }
  \fill[pink!80!red] (-0.13,-0.17) -- (0.13,-0.17) -- (0,-0.32) -- cycle;
  \draw[thick] (0,-0.32) -- (0,-0.42) (0,-0.42) arc (0:-150:0.17) (0,-0.42) arc (180:330:0.17);
  % Whiskers
  \foreach \s in {-1,1} \foreach \a in {-12,0,12}
    \draw[gray!60!black] (\s*0.35,-0.35) -- ++({90-\s*90+\s*\a}:0.95);
\end{tikzpicture}
