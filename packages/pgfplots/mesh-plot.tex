% Name: 3D mesh plot
% Description: The saddle z = x^2 - y^2 as a wireframe mesh, its lines colored by height.
% Tags: 3D, mesh, wireframe, saddle

\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

\begin{tikzpicture}
  \begin{axis}[
    width=9cm, view={35}{30}, colormap/viridis,
    xlabel=$x$, ylabel=$y$, zlabel=$z$,
    xtick={-2,0,2}, ytick={-2,0,2}, ztick={-4,0,4},
  ]
    \addplot3[mesh, thick, domain=-2:2, domain y=-2:2, samples=21] {x^2 - y^2};
  \end{axis}
\end{tikzpicture}
